\clearpage\section{Аналитическая часть}
\label{sec:analysis}
Постановка задачи: дана последовательность целых чисел $a=\\=(a_0,a_1,\ldots,a_n)$, где $n$ — длина последовательности, и искомое значение $T$. Требуется найти такой индекс $i$ последовательности чисел $a$, что $a_i = T$.

\subsection{Метод полного перебора (Линейный поиск)}
Алгоритм последовательного перебора выполняет один линейный проход по элементам несортированной последовательности:
\begin{itemize}
\item инициализировать счётчик $i \leftarrow 0$;
\item для каждого элемента $a_i$:
  \begin{itemize}
  \item если $a_i = T$, вернуть $i$;
  \item выполнить $i \leftarrow i + 1$;
  \end{itemize}
\item вернуть $-1$.
\end{itemize}
Алгоритм оперирует базовыми операциями индексации и проверкой равенства значений.

\subsection{Рекурсивный бинарный поиск}
Алгоритм. Рекурсивная функция поиска принимает на вход отсортированную по возрастанию последовательность чисел $A$, искомое значение $T$, а также левую ($l$) и правую ($r$) границы поиска:
\begin{itemize}
\item если $l > r$, вернуть $-1$;
\item иначе:
  \begin{itemize}
  \item инициализировать $m \leftarrow \lfloor\frac{l+r}{2}\rfloor$;
  \item если $T = a_m$, вернуть $m$;
  \item иначе если $T < a_m$, выполнить рекурсивный поиск для левой половины (вызвать функцию с границами $l, m-1$);
  \item иначе выполнить рекурсивный поиск для правой половины (вызвать функцию с границами $m+1, r$).
  \end{itemize}
\end{itemize}

\subsection{Итеративный бинарный поиск №1 (с двумя сравнениями)}
Классический итеративный подход сравнивает ключ $T$ со средним элементом $a_m$ отсортированного массива на каждой итерации. Если они равны, поиск завершается, в противном случае область сужается влево или вправо:
\begin{itemize}
\item инициализировать $l \leftarrow 0$, $r \leftarrow n-1$;
\item пока $l \le r$:
  \begin{itemize}
  \item инициализировать $m \leftarrow \lfloor\frac{l+r}{2}\rfloor$;
  \item если $T = a_m$, вернуть $m$;
  \item иначе если $T < a_m$, выполнить $r \leftarrow m-1$;
  \item иначе выполнить $l \leftarrow m+1$;
  \end{itemize}
\item вернуть $-1$.
\end{itemize}

\subsection{Итеративный бинарный поиск №2 (с одним сравнением)}
Альтернативный итеративный алгоритм откладывает проверку на точное равенство до завершения основного цикла сужения диапазона:
\begin{itemize}
\item инициализировать $l \leftarrow 0$, $r \leftarrow n-1$;
\item пока $l < r$:
  \begin{itemize}
  \item инициализировать $m \leftarrow l + \lfloor\frac{r-l}{2}\rfloor$;
  \item если $a_m > T$, выполнить $r \leftarrow m-1$;
  \item иначе выполнить $l \leftarrow m$;
  \end{itemize}
\item если $l \le r$ и $a_l = T$, вернуть $l$;
\item вернуть $-1$.
\end{itemize}

\conclusion{Описана задача поиска элемента в последовательности чисел, разработаны 4 алгоритма (линейный, рекурсивный и два итеративных бинарных поиска) для решения поставленной задачи.}

int linearSearch(const std::vector<int>& arr, int target, long long& comparisonCount) {
    comparisonCount = 0;
    for (size_t i = 0; i < arr.size(); ++i) {
        comparisonCount++;
        if (arr[i] == target) {
            return static_cast<int>(i);
        }
    }
    return -1;
}
\end{lstlisting}

\medskip
\noindent\begin{minipage}{\linewidth}
\begin{lstlisting}[caption={Реализация нерекурсивного алгоритма бинарного поиска с двумя сравнениями},label={lst:two}]
int binarySearchIterativeTwoCmp(const std::vector<int>& arr, int target, long long& comparisonCount) {
    comparisonCount = 0;
    int left = 0;
    int right = static_cast<int>(arr.size()) - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        comparisonCount++;
        if (arr[mid] == target) {
            return mid;
        }
        comparisonCount++;
        if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    return -1;
}
\end{lstlisting}
\end{minipage}

\clearpage
\noindent\begin{minipage}{\linewidth}
\begin{lstlisting}[caption={Реализация алгоритма нерекурсивного алгоритма бинарного поиска с одним сравнением},label={lst:one}]
int binarySearchIterativeOneCmp(const std::vector<int>& arr, int target, long long& comparisonCount) {
    comparisonCount = 0;
    int left = 0;
    int right = static_cast<int>(arr.size()) - 1;
    while (left < right) {
        int mid = left + (right - left + 1) / 2;
        comparisonCount++;
        if (arr[mid] > target) {
            right = mid - 1;
        } else {
            left = mid;
        }
    }
    if (left <= right && !arr.empty()) {
        comparisonCount++;
        if (arr[left] == target) {
            return left;
        }
    }
    return -1;
}
\end{lstlisting}
\end{minipage}

\medskip
\noindent\begin{minipage}{\linewidth}
\begin{lstlisting}[caption={Реализация рекурсивного алгоритма бинарного поиска},label={lst:recursive}]
int binarySearchRecursiveHelper(const std::vector<int>& arr, int target, int left, int right, long long& comparisonCount) {
    if (left > right) {
        return -1;
    }
    int mid = left + (right - left) / 2;
    comparisonCount++;
    if (arr[mid] == target) {
        return mid;
    }
    comparisonCount++;
    if (arr[mid] < target) {
        return binarySearchRecursiveHelper(arr, target, mid + 1, right, comparisonCount);
    } else {
        return binarySearchRecursiveHelper(arr, target, left, mid - 1, comparisonCount);
    }
}

int binarySearchRecursive(const std::vector<int>& arr, int target, long long& comparisonCount) {
    comparisonCount = 0;
    return binarySearchRecursiveHelper(arr, target, 0, static_cast<int>(arr.size()) - 1, comparisonCount);
}
\end{lstlisting}
\end{minipage}

\clearpage\section{Исследовательская часть}
\label{sec:research}

\subsection{Теоретическая оценка затрачиваемой памяти}
В алгоритме полного перебора и нерекурсивных бинарных поисках не задействуются дополнительные динамические структуры данных, используются только скалярные переменные, объем которых не зависит от входного размера $n$. Следовательно, оценка затрачиваемой памяти для них составляет $O(1)$.

В алгоритме рекурсивного бинарного поиска на каждый уровень вложенности вызова расходуется память системного стека под локальные переменные. В лучшем случае (элемент найден на первой итерации) глубина рекурсии равна 1, затраты памяти $O(1)$. В худшем случае (элемент отсутствует) глубина вызовов достигает $\log_2 n$, что дает оценку затрачиваемой памяти $O(\log n)$.

\subsection{Оценка вычислительной сложности алгоритмов как функции от числа сравнений}
Пусть количество элементов последовательности равно $n$. Для работы алгоритмов бинарного поиска входной массив должен быть предварительно отсортирован по возрастанию.

\subsubsection{Метод полного перебора}
Худшим случаем является отсутствие искомого элемента в последовательности (либо его нахождение в самом конце), тогда сложность по числу сравнений составляет $O(n)$. Лучший случай — нахождение элемента на первой позиции ($O(1)$).

\subsubsection{Итеративный бинарный поиск с двумя сравнениями}
Лучший случай — элемент находится ровно в центре массива на первой итерации ($O(1)$). Худший случай — элемент отсутствует, сложность составляет $O(\log n)$.

\subsubsection{Итеративный бинарный поиск с одним сравнением}
Алгоритм совершает фиксированное число шагов сужения диапазона, а проверка равенства вынесена за цикл. Вычислительная сложность в общем случае составляет $O(\log n)$.

\subsubsection{Рекурсивный бинарный поиск}
Аналогично итеративным аналогам, сложность по числу ключевых сравнений составляет $O(1)$ в лучшем случае и $O(\log n)$ в худшем случае.
