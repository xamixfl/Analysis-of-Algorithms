cmake_minimum_required(VERSION 3.16)
project(reverse_number LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(algorithms algorithms.cpp)
add_executable(reverse_number main.cpp)
target_link_libraries(reverse_number PRIVATE algorithms)
add_executable(test_algorithms tests/test_algorithms.cpp)
target_link_libraries(test_algorithms PRIVATE algorithms)
enable_testing()
add_test(NAME algorithms COMMAND test_algorithms)
