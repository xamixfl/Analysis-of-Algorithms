cmake_minimum_required(VERSION 3.16)
project(matrix_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(matrix_lab main.cpp algorithms.cpp)
if(MSVC)
  target_compile_options(matrix_lab PRIVATE /W4 /O2 /utf-8)
else()
  target_compile_options(matrix_lab PRIVATE -Wall -Wextra -Wpedantic -O2)
endif()
enable_testing()
add_test(NAME correctness COMMAND matrix_lab --test)
